function(tv_add_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE TokenViewerCore Qt5::Test)
    target_compile_definitions(${name} PRIVATE TV_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
    add_test(NAME ${name} COMMAND ${name})
    # Without a console window Qt Test writes to OutputDebugString, so ctest would capture nothing.
    set_tests_properties(${name} PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_FORCE_STDERR_LOGGING=1"
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${TV_QT_BIN_DIR_NATIVE}")
endfunction()
